function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet() .getSheetByName("Match_Results"); // sheet name // Create sheet if not exists if (!sheet) { sheet = SpreadsheetApp.getActiveSpreadsheet().insertSheet("Match_Results"); sheet.appendRow(["Date","Round","Tournament","GFC_Player", "GFC_ID","Opponent","GFC_Score","Opp_Score","Result","Submitted_At"]); } var data = JSON.parse(e.postData.contents); var rows = data.matches; var now = new Date().toISOString(); rows.forEach(function(m) { sheet.appendRow([ m.date, m.round, m.tournament, m.gfc_player, m.gfc_id, m.opponent, m.gfc_score, m.opp_score, m.result, now ]); }); return ContentService .createTextOutput(JSON.stringify({status:"ok", rows: rows.length})) .setMimeType(ContentService.MimeType.JSON); }
